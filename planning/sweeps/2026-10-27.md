The exact Playwright procedure was saved as `qa/check.mjs` in that project:

```javascript
import assert from "node:assert/strict";
import { chromium } from "@playwright/test";

const url = process.argv[2];
const retry = process.argv.includes("--retry");
assert.ok(url, "usage: node qa/check.mjs <url> [--retry]");

const browser = await chromium.launch();
try {
  const page = await browser.newPage();
  const pageErrors = [];
  const wasmResponses = [];
  page.on("pageerror", (error) => pageErrors.push(String(error)));
  page.on("response", (response) => {
    if (new URL(response.url()).pathname.endsWith(".wasm")) {
      wasmResponses.push({ status: response.status(), url: response.url() });
    }
  });

  let failedWasmUrl;
  if (retry) {
    await page.route("**/*.wasm", async (route) => {
      failedWasmUrl = route.request().url();
      await route.fulfill({ status: 503, body: "intentional first-WASM failure" });
    });
  }

  await page.goto(url);
  const button = page.locator("#download");
  const status = page.locator("#status");
  await button.click();

  if (retry) {
    await page.waitForFunction(
      () => document.querySelector("#status").textContent.startsWith("Download failed:"),
      null,
      { timeout: 195_000 },
    );
    assert.equal(
      await status.textContent(),
      "Download failed: Could not initialize the Autonomi WASM core",
    );
    assert.equal(await button.isEnabled(), true);
    assert.ok(failedWasmUrl?.endsWith(".wasm"));
    await page.unroute("**/*.wasm");
    await button.click();
  }

  await page.waitForFunction(
    () => document.querySelector("#status").textContent === "Downloaded 138931 bytes.",
    null,
    { timeout: 195_000 },
  );
  await page.waitForFunction(() => document.querySelector("#preview").naturalWidth > 0);
  const image = await page.locator("#preview").evaluate(async (element) => {
    const bytes = await (await fetch(element.src)).arrayBuffer();
    const digest = await crypto.subtle.digest("SHA-256", bytes);
    return {
      bytes: bytes.byteLength,
      width: element.naturalWidth,
      height: element.naturalHeight,
      sha256: [...new Uint8Array(digest)]
        .map((byte) => byte.toString(16).padStart(2, "0"))
        .join(""),
    };
  });

  assert.deepEqual(image, {
    bytes: 138931,
    width: 782,
    height: 1600,
    sha256: "98f657d987d339c302295e79907e7a4abc1564bd6b42300ea8d59ccd2148fd17",
  });
  assert.equal(await button.isEnabled(), true);
  assert.deepEqual(pageErrors, []);
  assert.equal(wasmResponses.at(-1)?.status, 200);
  console.log(JSON.stringify({
    url,
    retry,
    browser: browser.version(),
    platform: process.platform,
    arch: process.arch,
    failedWasmUrl,
    finalStatus: await status.textContent(),
    image,
    wasmResponses,
    pageErrors,
    buttonRestored: await button.isEnabled(),
  }, null, 2));
} finally {
  await browser.close();
}
```
